# Benchmarks: one executable per allocator backend.
find_package(benchmark CONFIG QUIET)
if(NOT benchmark_FOUND)
  FetchContent_Declare(benchmark URL https://github.com/google/benchmark/archive/refs/tags/v1.9.1.tar.gz
                       URL_HASH SHA256=32131c08ee31eeff2c8968d7e874f3cb648034377dfc32a4c377fa8796d84981)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(benchmark)
endif()

find_package(mimalloc CONFIG QUIET)
find_library(JEMALLOC_LIB NAMES jemalloc HINTS /opt/homebrew/opt/jemalloc/lib /usr/local/opt/jemalloc/lib)

set(HW_BACKENDS system heapwise)
if(mimalloc_FOUND)
  list(APPEND HW_BACKENDS mimalloc)
endif()
if(JEMALLOC_LIB)
  list(APPEND HW_BACKENDS jemalloc)
endif()
message(STATUS "heapwise benchmark backends: ${HW_BACKENDS}")

add_executable(hw_tracecompile tracecompile.cpp)
target_compile_options(hw_tracecompile PRIVATE ${HW_WARNINGS})

foreach(b IN LISTS HW_BACKENDS)
  string(TOUPPER ${b} B)
  foreach(kind workloads micro)
    if(kind STREQUAL "workloads")
      set(t bench_${b})
    else()
      set(t micro_${b})
    endif()
    add_executable(${t} ${kind}.cpp)
    target_compile_definitions(${t} PRIVATE HW_BACKEND_${B}=1)
    target_compile_options(${t} PRIVATE ${HW_WARNINGS})
    target_link_libraries(${t} PRIVATE Threads::Threads)
    if(kind STREQUAL "micro")
      target_link_libraries(${t} PRIVATE benchmark::benchmark)
    endif()
    if(b STREQUAL "heapwise")
      target_link_libraries(${t} PRIVATE heapwise)
    elseif(b STREQUAL "mimalloc")
      # Homebrew ships mimalloc-static; Debian/Ubuntu's package only defines the shared target.
      if(TARGET mimalloc-static)
        target_link_libraries(${t} PRIVATE mimalloc-static)
      else()
        target_link_libraries(${t} PRIVATE mimalloc)
      endif()
    elseif(b STREQUAL "jemalloc")
      target_link_libraries(${t} PRIVATE ${JEMALLOC_LIB})
    endif()
  endforeach()
endforeach()

if(APPLE)
  add_library(hwtracer SHARED ${PROJECT_SOURCE_DIR}/tools/tracer/tracer.c)
  target_compile_options(hwtracer PRIVATE -Wall -Wextra -Werror)
  set_target_properties(hwtracer PROPERTIES PREFIX "lib" OUTPUT_NAME hwtracer)
endif()

# Ablations: heapwise with one optimisation disabled, for bench/scripts/ablate.sh.
foreach(abl HYSTERESIS DIRECT_TSD)
  string(TOLOWER ${abl} name)
  add_library(heapwise_no_${name} STATIC ${PROJECT_SOURCE_DIR}/src/heapwise.cpp ${PROJECT_SOURCE_DIR}/src/os.cpp)
  target_include_directories(heapwise_no_${name} PUBLIC ${PROJECT_SOURCE_DIR}/include PRIVATE ${PROJECT_SOURCE_DIR}/src)
  target_compile_definitions(heapwise_no_${name} PRIVATE HW_ABLATE_${abl}=1)
  target_compile_options(heapwise_no_${name} PRIVATE ${HW_WARNINGS} -fno-exceptions -fno-rtti)
  add_executable(ablate_no_${name} workloads.cpp)
  target_compile_definitions(ablate_no_${name} PRIVATE HW_BACKEND_HEAPWISE=1)
  target_compile_options(ablate_no_${name} PRIVATE ${HW_WARNINGS})
  target_link_libraries(ablate_no_${name} PRIVATE heapwise_no_${name} Threads::Threads)
endforeach()
